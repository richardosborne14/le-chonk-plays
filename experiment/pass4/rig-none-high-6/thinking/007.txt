dogStep algorithm:
```js
const dogStep = (fromCol, fromRow, toCol, toRow) => {
  const dCol = toCol - fromCol;
  const dRow = toRow - fromRow;
  const axes = Math.abs(dCol) >= Math.abs(dRow) ? ["across", "updown"] : ["updown", "across"];
  for (const axis of axes) {
    if (axis === "across" && dCol !== 0) {
      const nextCol = fromCol + (dCol > 0 ? 1 : -1);
      if (!isCrate(nextCol, fromRow)) return [nextCol, fromRow];
    }
    if (axis === "updown" && dRow !== 0) {
      const nextRow = fromRow + (dRow > 0 ? 1 : -1);
      if (!isCrate(fromCol, nextRow)) return [fromCol, nextRow];
    }
  }
  return [fromCol, fromRow];
};
```
